def audit_process(proc_nombre, anio, periodo, idx):
    seed_val = anio * 10 + (1 if periodo == "I" else 2) + idx * 7
    issues = []

    # Generar las 3 pruebas
    try:
        p_dia1 = get_dia1_100_aptitud_humanidades(proc_nombre, anio, periodo, seed_val)
    except Exception as e:
        issues.append(f"CRITICAL ERROR in Dia 1 generation: {e}")
        p_dia1 = []

    try:
        p_dia2 = get_dia2_matematica(proc_nombre, anio, periodo, seed_val)
    except Exception as e:
        issues.append(f"CRITICAL ERROR in Dia 2 generation: {e}")
        p_dia2 = []

    try:
        p_dia3 = get_dia3_fisica_quimica(proc_nombre, anio, periodo, seed_val)
    except Exception as e:
        issues.append(f"CRITICAL ERROR in Dia 3 generation: {e}")
        p_dia3 = []

    all_questions = [("Día 1", p) for p in p_dia1] + [("Día 2", p) for p in p_dia2] + [("Día 3", p) for p in p_dia3]

    for dia_label, q in all_questions:
        q_num = q.get("num", 0)
        materia = q.get("materia", "")
        opciones = q.get("opciones", {})
        clave = q.get("clave", "")
        enunciado = q.get("enunciado", "")
        solucion = q.get("solucion", "")

        # 1. Verificar número de opciones
        if len(opciones) != 5:
            issues.append(f"[{dia_label} P{q_num}] Opciones distintas de 5: {len(opciones)}")

        # 2. Verificar opciones duplicadas
        vals = list(opciones.values())
        if len(set(vals)) != len(vals):
            issues.append(f"[{dia_label} P{q_num} ({materia})] Opciones duplicadas detectadas: {opciones}")

        # 3. Verificar que la clave existe en opciones
        if clave not in opciones:
            issues.append(f"[{dia_label} P{q_num}] Clave '{clave}' no existe en opciones {list(opciones.keys())}")

        # 4. Verificar contenido no vacío
        if not enunciado or len(enunciado.strip()) < 10:
            issues.append(f"[{dia_label} P{q_num}] Enunciado vacío o demasiado corto")
        if not solucion or len(solucion.strip()) < 10:
            issues.append(f"[{dia_label} P{q_num}] Solución vacía o demasiado corta")

        # 5. Verificar coherencia textual en la solución respecto a la clave
        # Buscar si la solución menciona otra clave distinta o contradicción
        # E.g., si la clave es C pero la solución dice 'clave B' o 'opción B'
        match_clave = re.search(r"clave\s+es\s+([A-E])|opci[oó]n\s+([A-E])|respuesta\s+es\s+([A-E])", solucion, re.I)
        if match_clave:
            found_c = [g for g in match_clave.groups() if g]
            if found_c and found_c[0].upper() != clave.upper():
                issues.append(f"[{dia_label} P{q_num}] Posible contradicción: clave es '{clave}' pero solución menciona '{found_c[0]}'")

    return len(all_questions), issues

### 3. LOS TEMAS 'FIJOS' QUE SU HIJA DEBE DOMINAR (PROBABILIDAD 95% - 99%)

#### A. FÍSICA (20 Preguntas - Día 3)
1. **Estática (Fija 99.5%)**: Barra con pivote o apoyos en equilibrio. Tomar momentos (torques) en el punto donde se cruzan las fuerzas desconocidas para anularlas.
2. **Cinemática MRUV y Caída Libre (Fija 99.2%)**: Ecuaciones horarias de distancia, tiempo y altura máxima.
3. **Conservación de la Energía (Fija 98.8%)**: Energía cinética + potencial. Si hay fricción, la energía perdida es igual a fuerza de rozamiento por distancia.
4. **Dinámica Lineal (Fija 98.8%)**: Segunda ley de Newton (Fuerza neta = masa x aceleración).
5. **Hidrostática (Fija 97.2%)**: Principio de Arquímedes. Empuje hacia arriba igual al peso del líquido desalojado.
6. **Circuitos Eléctricos (Fija 97.2%)**: Resistencias en serie y paralelo, Ley de Ohm (V = I x R) y potencia disipada (P = I^2 x R).
7. **Fuerza Magnética de Lorentz (Fija 96.4%)**: Carga en movimiento dentro de un campo magnético. Trayectoria circular con radio R = mv / (qB).

#### B. QUÍMICA (20 Preguntas - Día 3)
1. **Estequiometría (Fija 99.0%)**: Reactivo limitante y rendimiento porcentual. Dividir moles dados entre su coeficiente; el menor manda.
2. **Estructura Atómica y Configuración (Fija 98.8%)**: Regla del serrucho (Aufbau) y números cuánticos (n, l, m, s).
3. **Tabla Periódica y Electronegatividad (Fija 98.7%)**: Variación de radio atómico y electronegatividad (el Flúor es el rey con 4.0).
4. **Gases Ideales (Fija 98.4%)**: Ecuación universal P.V = n.R.T y presiones parciales de Dalton. A Condiciones Normales, 1 mol = 22.4 L.
5. **Enlace Químico (Fija 98.5%)**: Enlaces sigma y pi en compuestos orgánicos e inorgánicos.
6. **Soluciones y Dilución (Fija 97.5%)**: Molaridad y conservación de soluto: C1 x V1 = C2 x V2.
7. **Equilibrio Químico (Fija 96.8%)**: Principio de Le Chatelier. Si se comprime el gas, corre hacia donde hay menos moléculas.
8. **Electroquímica (Fija 96.8%)**: Pila Daniell y Leyes de Faraday.

#### C. ARITMÉTICA (10 Preguntas - Día 2)
1. **Divisibilidad y Residuos (Fija 98.8%)**: Pequeño Teorema de Fermat. Si el divisor es primo (ej. 7), el exponente se reduce dividiéndolo entre 6.
2. **Teoría de Conjuntos (Fija 98.5%)**: Diagramas de Venn e Inclusión-Exclusión para 2 y 3 conjuntos.
3. **MCD y MCM (Fija 96.5%)**: Propiedad fundamental: A x B = MCD(A, B) x MCM(A, B).
4. **Estudio de Divisores (Fija 97.0%)**: Cantidad de divisores primos, compuestos y totales mediante descomposición canónica.
5. **Razones y Proporciones (Fija 98.0%)**: Serie de razones geométricas iguales.

#### D. ÁLGEBRA (10 Preguntas - Día 2)
1. **Productos Notables (Fija 98.9%)**: Identidades de Legendre y caso condicional (si a+b+c=0, a^3+b^3+c^3 = 3abc).
2. **Relaciones de Cardano-Viète (Fija 98.6%)**: Suma y producto de raíces en polinomios cúbicos y cuadráticos sin necesidad de resolver la ecuación.
3. **Leyes de Exponentes (Fija 98.2%)**: Ecuaciones exponenciales igualando bases primas.
4. **Matrices y Determinantes (Fija 97.8%)**: Propiedad ineludible: det(k.A) = k^n . det(A) para matriz de orden n.
5. **Inecuaciones Fraccionarias (Fija 97.0%)**: Método de puntos críticos en la recta real.

#### E. GEOMETRÍA (10 Preguntas - Día 2)
1. **Relaciones Métricas en Triángulos (Fija 99.1%)**: Pitágoras, teorema de la altura (h^2 = m.n) y producto de catetos = hipotenusa x altura.
2. **Ángulos en Triángulos (Fija 99.1%)**: Ángulo entre bisectrices interiores: x = 90° + A/2.
3. **Áreas Sombreadas (Fija 98.4%)**: Traslación de regiones simétricas y diferencia entre polígonos y círculos inscritos.
4. **Teorema de Poncelet (Fija 98.2%)**: Cateto + Cateto = Hipotenusa + 2 x Inradio.
5. **Geometría del Espacio y Euler (Fija 97.4%)**: Caras + Vértices = Aristas + 2. Volúmenes de cilindro, cono y esfera.

#### F. TRIGONOMETRÍA (10 Preguntas - Día 2)
1. **Razones Notables (Fija 99.0%)**: Triángulos 30°-60°, 45°-45°, 37°-53°.
2. **Arcos Compuestos y Arco Doble (Fija 98.5%)**: Fórmulas de seno y coseno de la suma, ángulo doble y degradación cuadrática.
3. **Circunferencia Trigonométrica (Fija 97.0%)**: Variación de líneas de seno y coseno en los 4 cuadrantes.
4. **Ley de Senos y Cosenos (Fija 97.5%)**: a/sen(A) = b/sen(B) = 2R y c^2 = a^2 + b^2 - 2ab cos(C).

#### G. RAZONAMIENTO MATEMÁTICO (35 Preguntas - Día 1)
1. **Sucesiones y Progresiones (Fija 99.5%)**: Progresión aritmética y término general.
2. **Certezas y Peor Escenario (Fija 98.9%)**: Ponerse siempre en el caso más desfavorable (sacar todo lo indeseado primero).
3. **Problemas de Edades (Fija 98.8%)**: La diferencia de edades entre dos personas nunca cambia en el tiempo.
4. **Operadores Matemáticos en Tablas (Fija 98.2%)**: Buscar elemento neutro 'e' e inverso 'a^-1'.
5. **Conteo de Figuras y Trazos (Fija 98.0%)**: Vértices impares de Euler (máximo 2 para recorrer sin levantar el lápiz).

            m = q.get('materia', 'General')
            topics[t]['count'] += 1
            topics[t]['materia'] = m
            topics[t]['arquetipos'].add(q.get('arquetipo', ''))

sorted_top = sorted(topics.items(), key=lambda x: x[1]['count'], reverse=True)
